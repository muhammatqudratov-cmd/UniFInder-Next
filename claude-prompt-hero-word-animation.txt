          charIndex.current = word.length;

        charIndex.current--;

          setWordIndex((prev) => (prev + 1) % WORDS.length);
        }
      }, deleting && charIndex.current === word.length ? 2000 : 50);
    }
    return () => clearTimeout(timeout);
  }, [text, deleting, wordIndex]);

  return text;
}
