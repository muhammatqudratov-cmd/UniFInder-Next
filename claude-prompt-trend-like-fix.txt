Replace with:
	const likeUniversityHandler = async (user: T, id: string) => {
		try {
			if (!id) return;
			if (!user._id) throw new Error(Message.NOT_AUTHENTICATED);
			// Execute like
			await likeTargetUniversity({ variables: { input: id } });
			// Optimistic local update — no refetch, no re-render of all cards
			setTrendUniversities((prev) =>
				prev.map((uni) => {
					if (uni._id !== id) return uni;
					const liked = uni.meLiked?.[0]?.myFavorite ?? false;
					return {
						...uni,
						universityLikes: liked
							? Math.max((uni.universityLikes ?? 1) - 1, 0)
							: (uni.universityLikes ?? 0) + 1,
						meLiked: [{ myFavorite: !liked }],
					};
				}),
			);
			await sweetTopSmallSuccessAlert('Success', 800);
		} catch (err: any) {
			console.log('ERROR, likeUniversityHandler', err);
			sweetMixinErrorAlert(err.message).then();
		}
	};
